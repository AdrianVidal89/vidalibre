{# Una portada de libro. Se usa igual en la rejilla de la biblioteca que en los carruseles de la portada, para que un libro se vea siempre igual. #} {% macro tarjeta(libro, leido=none, en_marcha=false) %}
  • {% if leido and leido > 0.01 %} {% endif %} {{ libro.titulo }} {{ libro.autores | join(', ') }} {% if en_marcha and leido %} {{ (leido * 100) | round | int }}% leído {% endif %} {% if not (en_marcha and leido) %} {# Fuera del enlace del libro: un enlace no puede ir dentro de otro. #} {% for formato in libro.formatos %}{{ formato }}{% endfor %} {% if libro.serie %}{{ libro.serie }} · {{ libro.indice_serie | round(1) }}{% endif %} {% endif %}
    • {% endmacro %} {% macro rejilla(libros, lecturas={}) %}
      {% for libro in libros %}{{ tarjeta(libro, lecturas.get(libro.id)) }}{% endfor %}
    {% endmacro %} {% macro carrusel(libros, clase="", lecturas={}) %}
      {% for libro in libros %} {{ tarjeta(libro, libro.fraccion if libro.fraccion is defined else lecturas.get(libro.id), en_marcha=(clase == 'en-marcha')) }} {% endfor %}
    {% endmacro %}
